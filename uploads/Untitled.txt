# ------------ HELPERS ------------
def pages_from_pdf(path):
    try:
        reader = PdfReader(path)
        return len(reader.pages)
    except Exception:
        return None


def parse_pages_input(pages_input, file_path=None):
    if not pages_input:
        return 1

    s = str(pages_input).strip().lower()

    # full/all
    if re.search(r"\b(full|all|entire|complete)\b", s):
        if file_path and file_path.lower().endswith(".pdf"):
            pdf_pages = pages_from_pdf(file_path)
            return pdf_pages if pdf_pages else 1
        return 1

    cleaned = re.sub(r"[^\d,\-]", "", s)
    if cleaned == "":
        return 1

    if "," not in cleaned and "-" not in cleaned:
        try:
            return max(1, int(cleaned))
        except Exception:
            return 1

    pages_set = set()
    for part in cleaned.split(","):
        if not part:
            continue
        if "-" in part:
            try:
                a, b = map(int, part.split("-"))
                if b < a:
                    a, b = b, a
                for pg in range(a, b + 1):
                    pages_set.add(pg)
            except Exception:
                continue
        else:
            try:
                pages_set.add(int(part))
            except Exception:
                continue

    return len(pages_set) if pages_set else 1

@app.route("/uploads/<path:filename>")
def uploaded_file(filename):
    # Serve files from the upload folder. Flask's send_from_directory handles safe headers,
    # but we still check that the file exists to return 404 if missing.
    upload_dir = os.path.abspath(app.config["UPLOAD_FOLDER"])
    target = os.path.abspath(os.path.join(upload_dir, filename))
    if not target.startswith(upload_dir):
        # prevent directory traversal
        abort(404)
    if not os.path.exists(target):
        abort(404)
    return send_from_directory(app.config["UPLOAD_FOLDER"], filename)

@app.route("/dashboard", methods=["GET", "POST"])
def dashboard():
    # student-only
    if "user_id" not in session or session["role"] != "student":
        return redirect(url_for("login"))
    completed_requests = PrintRequest.query.filter_by(   # ✅ SHOW FLASH WHEN PRINT COMPLETES
        user_id=session["user_id"],
        status="Completed",
        print_notified=False
    ).all()

    for req in completed_requests:
        flash(
            f"✅ Your print request #{req.id} has been completed!",
            "success"
        )
        req.print_notified = True

    if completed_requests:
        db.session.commit()

    if request.method == "POST":
        file = request.files.get("file")

        if file and file.filename != "":
            # Validate file extension
            if not allowed_file(file.filename):
                flash("Invalid file type! Allowed: PDF, PNG, JPG, JPEG", "danger")
                return redirect(url_for("dashboard"))

            # secure and unique filename: prefix with user id and timestamp to reduce collisions
            filename_raw = secure_filename(file.filename)
            # you can add timestamp or uuid if needed; keeping simpler:
            filename = f"user{session['user_id']}_{filename_raw}"

            os.makedirs(app.config["UPLOAD_FOLDER"], exist_ok=True)
            file_path = os.path.join(app.config["UPLOAD_FOLDER"], filename)
            try:
                file.save(file_path)
            except Exception:
                flash("Failed to save uploaded file.", "danger")
                return redirect(url_for("dashboard"))

            # form fields
            paper_size = request.form.get("paper_size")
            orientation = request.form.get("orientation")
            color_mode = request.form.get("color_mode")
            print_quality = request.form.get("print_quality")
            finishing = request.form.get("finishing")
            sidedness = request.form.get("sidedness")
            pages_input = request.form.get("pages", "1")

            try:
                copies = max(1, int(request.form.get("copies", 1)))
            except Exception:
                copies = 1

            page_count = parse_pages_input(pages_input, file_path=file_path)

            # ------ COSTING LOGIC ------
            color = (color_mode or "").lower()
            rate = 5 if "color" in color else 2

            side = (sidedness or "").lower()
            if "back" in side:
                sheets_per_copy = (page_count + 1) // 2
            else:
                sheets_per_copy = page_count

            printing_cost = sheets_per_copy * copies * rate

            fin = (finishing or "").lower()
            if "soft" in fin:
                finishing_cost = 15 * copies
            elif "spiral" in fin:
                finishing_cost = 30 * copies
            else:
                finishing_cost = 0

            total_cost = printing_cost + finishing_cost

            # Save request
            req = PrintRequest(
                user_id=session["user_id"],
                filename=filename,
                paper_size=paper_size,
                orientation=orientation,
                color_mode=color_mode,
                print_quality=print_quality,
                finishing=finishing,
                pages=pages_input,
                copies=copies,
                sidedness=sidedness,
                cost=total_cost,
            )

            db.session.add(req)
            db.session.commit()

            flash(f"Print Request Submitted! Total Cost: ₹{total_cost}")
            return redirect(url_for("dashboard"))
        else:
            flash("No file selected.", "warning")
            return redirect(url_for("dashboard"))

    requests = PrintRequest.query.filter_by(user_id=session["user_id"]).all()
    return render_template("dashboard.html", requests=requests)

@app.route("/staff")
def staff_panel():
    if "user_id" not in session or session["role"] != "staff":
        return redirect(url_for("staff_login"))

    requests = (
        PrintRequest.query
        .filter_by(visible_to_staff=True)
        .order_by(
            PrintRequest.payment_proof.is_(None),
            PrintRequest.created_at.desc()
        )
        .all()
    )

    today = datetime.now().date()
    yesterday = today - timedelta(days=1)

    for r in requests:
        if r.created_at:
            d = r.created_at.date()
            if d == today:
                r.date_label = "Today"
            elif d == yesterday:
                r.date_label = "Yesterday"
            else:
                r.date_label = r.created_at.strftime("%d-%m-%Y")
        else:
            r.date_label = "-"

    return render_template("staff.html", requests=requests)

@app.route("/delete/<int:req_id>", methods=["POST"])
def delete_request(req_id):
    if "user_id" not in session or session["role"] != "student":
        return redirect(url_for("login"))

    req = PrintRequest.query.get_or_404(req_id)
    if req.user_id != session["user_id"]:
        return redirect(url_for("dashboard"))

    # delete uploaded files (main file + proof) if present
    try:
        if req.filename:
            fp = os.path.join(app.config["UPLOAD_FOLDER"], req.filename)
            if os.path.exists(fp):
                os.remove(fp)
        if req.payment_proof:
            pp = os.path.join(app.config["UPLOAD_FOLDER"], req.payment_proof)
            if os.path.exists(pp):
                os.remove(pp)
    except Exception:
        # don't block deletion on file removal error
        pass

    db.session.delete(req)
    db.session.commit()

    return redirect(url_for("dashboard"))


@app.route("/upload_payment_proof/<int:req_id>", methods=["GET", "POST"])
def upload_payment_proof(req_id):
    if "user_id" not in session or session["role"] != "student":
        return redirect(url_for("login"))

    req = PrintRequest.query.get_or_404(req_id)
    if req.user_id != session["user_id"]:
        return redirect(url_for("dashboard"))

    if request.method == "POST":
        file = request.files.get("proof")
        if file and file.filename:
            # Validate proof file type
            if not allowed_file(file.filename):
                flash("Invalid proof file type! Allowed: PDF, PNG, JPG, JPEG", "warning")
                return redirect(url_for("upload_payment_proof", req_id=req.id))

            proof_name_raw = secure_filename(file.filename)
            filename = secure_filename(f"proof_{req.id}_{proof_name_raw}")
            path = os.path.join(app.config["UPLOAD_FOLDER"], filename)

                file.save(path)
            except Exception:
                flash("Failed to save proof file.", "danger")
                return redirect(url_for("upload_payment_proof", req_id=req.id))

            req.payment_proof = filename
            req.payment_status = "Paid"
            db.session.commit()

            flash("Payment proof uploaded successfully.")
            return redirect(url_for("dashboard"))
        else:
            flash("No proof file selected.", "warning")

    upi_link = f"upi://pay?pa={UPI_ID}&pn={UPI_NAME}&am={req.cost}&cu=INR&tn=QuickPrint%20Order%20{req.id}"
    return render_template("upload_payment_proof.html", req=req, upi_link=upi_link)
